If no ReplicaSet is initialized (manual monkey patch):
# From a terminal inside the mongo container, run the following commands:

# Start the MongoDB shell
mongosh

# Show the parsed command line options for the MongoDB server
db.adminCommand({ getCmdLineOpts: 1 }).parsed

# Initialize a new replica set named "rs0" with a single member
rs.initiate({
  _id: "rs0",
  members: [
    { _id: 0, host: "mongo:27017" }
  ]
})

# Check the status of the replica set
rs.status()
